        let imgElement = null;

        function resizeCanvas() {
            const rect = container.getBoundingClientRect();
            canvas.width = rect.width;
            canvas.height = rect.height;
            renderImage();
        }

        function renderImage() {
            ctx.clearRect(0, 0, canvas.width, canvas.height);
            if (!imgElement || !imgElement.complete || imgElement.naturalWidth === 0) {
                return;
            }
            const w = imgElement.naturalWidth;
            const h = imgElement.naturalHeight;
            const cw = canvas.width;
            const ch = canvas.height;
            const scale = state.zoom;
            const sw = w * scale;
            const sh = h * scale;
            const sx = (cw - sw) / 2 + state.panX;
            const sy = (ch - sh) / 2 + state.panY;
            ctx.imageSmoothingEnabled = true;
            ctx.imageSmoothingQuality = 'high';
            ctx.drawImage(imgElement, sx, sy, sw, sh);
        }

        function fitImage() {
            if (!imgElement) return;
            const cw = canvas.width;
            const ch = canvas.height;
            const iw = imgElement.naturalWidth;
            const ih = imgElement.naturalHeight;
            const scaleX = cw / iw;
            const scaleY = ch / ih;
            state.zoom = Math.min(scaleX, scaleY) * 0.92;
            state.panX = 0;
            state.panY = 0;
            renderImage();
        }

        // ---- 鼠标事件 ----
        let mouseDown = false;
        let lastMX = 0,
            lastMY = 0;

        canvas.addEventListener('mousedown', (e) => {
            if (!imgElement) return;
            mouseDown = true;
            lastMX = e.clientX;
            lastMY = e.clientY;
            state.isDragging = true;
            state.startPanX = state.panX;
            state.startPanY = state.panY;
            canvas.style.cursor = 'grabbing';
        });

        window.addEventListener('mousemove', (e) => {
            if (!mouseDown || !imgElement) return;
            const dx = e.clientX - lastMX;
            const dy = e.clientY - lastMY;
            state.panX = state.startPanX + dx;
            state.panY = state.startPanY + dy;
            renderImage();
        });

        window.addEventListener('mouseup', () => {
            if (mouseDown) {
                mouseDown = false;
                state.isDragging = false;
                canvas.style.cursor = 'grab';
            }
        });

        // ---- 缺陷小图生成（模拟） ----
        function generateDefectThumbnails() {
            const count = Math.floor(Math.random() * 8) + 3; // 3~10个
            const defectNames = ['隐裂', '崩边', '缺口', '脏污', '划伤', '穿孔', '叠片', '线痕'];
            const colors = ['#ff6b6b', '#ff9f4a', '#4a9eff', '#00d4aa', '#8fa4c8', '#ffd93d', '#a29bfe', '#fd79a8'];
            const list = [];
            for (let i = 0; i < count; i++) {
                const name = defectNames[i % defectNames.length];
                const color = colors[i % colors.length];
                const size = 80;
                const c = document.createElement('canvas');
                c.width = size;
                c.height = size;
                const cx = c.getContext('2d');
                cx.fillStyle = color;
                cx.shadowColor = 'rgba(0,0,0,0.5)';
                cx.shadowBlur = 8;
                if (Math.random() > 0.5) {
                    const x = 10 + Math.random() * 30;
                    const y = 10 + Math.random() * 30;
                    const w = 20 + Math.random() * 40;
                    const h = 20 + Math.random() * 40;
                    cx.fillRect(x, y, w, h);
                } else {
                    cx.beginPath();
                    const cx2 = 40 + (Math.random() - 0.5) * 30;
                    const cy = 40 + (Math.random() - 0.5) * 30;
                    const rx = 15 + Math.random() * 25;
                    const ry = 15 + Math.random() * 25;
                    cx.ellipse(cx2, cy, rx, ry, 0, 0, Math.PI * 2);
                    cx.fill();
                }
                cx.shadowBlur = 0;
                cx.strokeStyle = '#fff';
                cx.lineWidth = 1;
                cx.strokeRect(0, 0, size, size);
                const dataUrl = c.toDataURL('image/png');
                list.push({
                    name: name + ' #' + (i + 1),
                    desc: '面积: ' + (10 + Math.random() * 90).toFixed(0) + ' px²',
                    img: dataUrl
                });
            }
            state.defectImages = list;
            renderDefectList();
        }

        }

        function refreshStats() {
            const total = Math.floor(1200 + Math.random() * 300);
            const pass = Math.floor(total * (0.85 + Math.random() * 0.12));
            const fail = total - pass;
            const yieldRate = (pass / total * 100).toFixed(1) + '%';
            $('statTotal').textContent = total.toLocaleString();
            $('statPass').textContent = pass.toLocaleString();
            $('statFail').textContent = fail.toLocaleString();
            $('statYield').textContent = yieldRate;

            const defects = [
                { label: '隐裂', max: 50 },
                { label: '崩边', max: 35 },
                { label: '缺口', max: 25 },
                { label: '脏污', max: 18 },
                { label: '划伤', max: 12 },
            ];
            const rows = document.querySelectorAll('.defect-bar-row');
            defects.forEach((d, idx) => {
                if (rows[idx]) {
                    const val = Math.floor(Math.random() * d.max) + 1;
                    const pct = Math.min(100, (val / 50) * 100);
                    const fill = rows[idx].querySelector('.bar-fill');
                    const valueSpan = rows[idx].querySelector('.bar-value');
                    if (fill) fill.style.width = pct + '%';
                    if (valueSpan) valueSpan.textContent = val;
                }
            });
        }

        let selectedPath = [];

        function buildTreeData(obj, path = []) {
            const result = [];
            for (const key of Object.keys(obj)) {
                const val = obj[key];
                const isLeaf = val && typeof val === 'object' && val.Type !== undefined;
                const node = {
                    key: key,
                    path: [...path, key],
                    isLeaf: isLeaf,
                    children: isLeaf ? [] : buildTreeData(val, [...path, key]),
                    data: isLeaf ? val : null,
                };
                result.push(node);
            }
            return result;
        }

        function renderTreeForCategory(category) {
            const categoryData = state.params[category];
            if (!categoryData) {
                paramTree.innerHTML = `<div style="padding:16px;color:var(--text-muted);">未找到大类: ${category}</div>`;
                paramDetail.innerHTML = `<div class="empty-state"><p>无数据</p></div>`;
                return;
            }
            const treeData = buildTreeData(categoryData, [category]);
            paramTree.innerHTML = '';
            treeData.forEach(node => renderTreeNode(node, paramTree, 0, true));
            paramTree.querySelectorAll('.tree-node .children').forEach(el => el.classList.add('open'));
            paramTree.querySelectorAll('.tree-node .arrow').forEach(el => el.classList.add('open'));
            const firstLabel = paramTree.querySelector('.tree-node .node-label');
            if (firstLabel) firstLabel.click();
        }

        function renderTreeNode(node, parentEl, depth, isRoot = false) {
            const div = document.createElement('div');
            div.className = 'tree-node';
            const label = document.createElement('div');
            label.className = 'node-label';
            label.style.paddingLeft = (12 + depth * 16) + 'px';
            label.dataset.path = JSON.stringify(node.path);

            if (!node.isLeaf) {
                const arrow = document.createElement('span');
                arrow.className = 'arrow';
                arrow.textContent = '▶';
                label.appendChild(arrow);
                const icon = document.createElement('span');
                icon.className = 'icon';
                icon.textContent = '📁';
                label.appendChild(icon);
            } else {
                const arrow = document.createElement('span');
                arrow.className = 'arrow';
                arrow.style.visibility = 'hidden';
                label.appendChild(arrow);
                const icon = document.createElement('span');
                icon.className = 'icon';
                icon.textContent = '📄';
                label.appendChild(icon);
            }

            const nameSpan = document.createElement('span');
            nameSpan.className = 'name';
            nameSpan.textContent = node.key;
            label.appendChild(nameSpan);

            if (!node.isLeaf) {
                const badge = document.createElement('span');
                badge.className = 'badge';
                badge.textContent = node.children.length;
                label.appendChild(badge);
            } else {
                const comp = node.data.Competence;
                const badge = document.createElement('span');
                badge.className = 'badge';
                badge.textContent = 'Lv' + comp;
                badge.style.color = ['#8fa4c8', '#00d4aa', '#4a9eff', '#ff9f4a'][comp] || '#8fa4c8';
                label.appendChild(badge);
            }

            div.appendChild(label);

            if (!node.isLeaf) {
                const childrenDiv = document.createElement('div');
                childrenDiv.className = 'children';
                node.children.forEach(child => renderTreeNode(child, childrenDiv, depth + 1));
                div.appendChild(childrenDiv);
                label.addEventListener('click', (e) => {
                    e.stopPropagation();
                    const isOpen = childrenDiv.classList.contains('open');
                    childrenDiv.classList.toggle('open');
                    label.querySelector('.arrow').classList.toggle('open');
                    selectNode(node.path, label);
                });
            } else {
                label.addEventListener('click', (e) => {
                    e.stopPropagation();
                    selectNode(node.path, label);
                });
            }
            parentEl.appendChild(div);
        }

        function selectNode(path, labelEl) {
            selectedPath = path;
            paramTree.querySelectorAll('.node-label.active').forEach(el => el.classList.remove('active'));
            if (labelEl) labelEl.classList.add('active');
            showParamDetail(path);
        }

        function showParamDetail(path) {
            if (!path || path.length === 0) {
                paramDetail.innerHTML = `<div class="empty-state"><svg viewBox="0 0 24 24"><path d="M12 2C6.48 2 2 6.48 2 12s4.48 10 10 10 10-4.48 10-10S17.52 2 12 2zm0 18c-4.41 0-8-3.59-8-8s3.59-8 8-8 8 3.59 8 8-3.59 8-8 8zm-1-13h2v6h-2zm0 8h2v2h-2z"/></svg><p>从左侧选择参数分组</p></div>`;
                return;
            }
            let obj = state.params;
            for (const key of path) {
                if (obj && obj[key] !== undefined) obj = obj[key];
                else { obj = null; break; }
            }
            if (!obj || typeof obj !== 'object' || obj.Type !== undefined) {
                const parentPath = path.slice(0, -1);
                let parentObj = state.params;
                for (const key of parentPath) {
                    if (parentObj && parentObj[key] !== undefined) parentObj = parentObj[key];
                    else { parentObj = null; break; }
                }
                if (parentObj && typeof parentObj === 'object') renderParamGroup(parentPath, parentObj);
                else paramDetail.innerHTML = `<div class="empty-state"><p>无参数</p></div>`;
                return;
            }
            renderParamGroup(path, obj);
        }

        function renderParamGroup(path, obj) {
            const name = path[path.length - 1] || '根';
            let html = `<div class="param-group-title">📂 ${name}</div>`;
            html += `<div class="param-group-desc">路径: ${path.join(' / ')}  ·  ${Object.keys(obj).length} 项</div>`;
            const items = [];
            for (const key of Object.keys(obj)) {
                const val = obj[key];
                if (val && typeof val === 'object' && val.Type !== undefined) items.push({ key, data: val, isLeaf: true });
                else if (val && typeof val === 'object') items.push({ key, data: val, isLeaf: false });
            }
            if (items.length === 0) {
                html += `<div class="text-muted" style="padding:12px;">此分组为空</div>`;
                paramDetail.innerHTML = html;
                return;
            }
            for (const item of items) {
                if (item.isLeaf) {
                    html += renderParamItem(item.key, item.data, path);
                } else {
                    const subPath = [...path, item.key];
                    html += `<div style="margin:8px 0 4px 0;padding:6px 10px;background:var(--bg-primary);border-radius:var(--radius-sm);border-left:2px solid var(--accent-cyan);">
                        <div style="display:flex;align-items:center;gap:6px;cursor:pointer;font-weight:500;color:var(--text-secondary);" onclick="toggleSubGroup(this)">
                            <span class="arrow-sub">▶</span> 📁 ${item.key}
                            <span style="font-size:11px;color:var(--text-muted);font-weight:400;">${Object.keys(item.data).length} 项</span>
                        </div>
                        <div class="sub-group" style="padding-left:16px;display:none;margin-top:4px;">
                    `;
                    const subItems = [];
                    for (const sk of Object.keys(item.data)) {
                        const sv = item.data[sk];
                        if (sv && typeof sv === 'object' && sv.Type !== undefined) subItems.push({ key: sk, data: sv });
                    }
                    if (subItems.length === 0) html += `<div class="text-muted" style="padding:4px 0;">此分组无直接参数</div>`;
                    else for (const si of subItems) html += renderParamItem(si.key, si.data, subPath);
                    html += `</div></div>`;
                }
            }
            paramDetail.innerHTML = html;
        }

        function renderParamItem(key, data, parentPath) {
            const comp = data.Competence ?? 0;
            const type = data.Type ?? 'string';
            const limit = data.Limit ?? '';
            const value = data.Value ?? '';
            const canEdit = comp <= state.currentUserLevel;
            const compLabel = ['操作员', '产线工程师', '系统工程师', '管理员'][comp] || '未知';
            const compColor = ['lv0', 'lv1', 'lv2', 'lv3'][comp] || 'lv0';
            let controlHtml = '';
            switch (type) {
                case 'int':
                case 'float': {
                    const isInt = type === 'int';
                    const step = isInt ? 1 : 0.1;
                    const min = Array.isArray(limit) ? limit[0] : '';
                    const max = Array.isArray(limit) ? limit[1] : '';
                    const val = value !== undefined && value !== '' ? value : (isInt ? 0 : 0.0);
                    controlHtml = `
                        <input type="number" step="${step}" value="${val}" ${!canEdit ? 'disabled' : ''}
                               data-path="${JSON.stringify([...parentPath, key])}"
                               data-type="${type}" data-min="${min}" data-max="${max}"
                               class="param-input" />
                        ${Array.isArray(limit) ? `<span class="range-hint">[${limit[0]} ~ ${limit[1]}]</span>` : ''}
                    `;
                    break;
                }
                case 'bool': {
                    const isOn = value === 'TRUE' || value === true;
                    controlHtml = `
                        <div class="bool-switch" data-path="${JSON.stringify([...parentPath, key])}" data-type="bool">
                            <div class="track ${isOn ? 'on' : ''}" onclick="toggleBool(this)">
                                <div class="thumb"></div>
                            </div>
                            <span class="bool-label">${isOn ? 'TRUE' : 'FALSE'}</span>
                        </div>
                    `;
                    break;
                }
                case 'string':
                    controlHtml = `
                        <input type="text" value="${value || ''}" ${!canEdit ? 'disabled' : ''}
                               data-path="${JSON.stringify([...parentPath, key])}" data-type="string"
                               class="param-input" />
                    `;
                    break;
                case 'string_enum':
                case 'int_enum':
                case 'float_enum': {
                    const options = Array.isArray(limit) ? limit : [];
                    controlHtml = `
                        <select ${!canEdit ? 'disabled' : ''}
                                data-path="${JSON.stringify([...parentPath, key])}" data-type="enum"
                                class="param-input">
                            ${options.map(opt => `<option value="${opt}" ${opt === value ? 'selected' : ''}>${opt}</option>`).join('')}
                        </select>
                    `;
                    break;
                }
                case 'directory':
                case 'path': {
                    controlHtml = `
                        <div style="display:flex;gap:4px;flex:1;max-width:280px;">
                            <input type="text" value="${value || ''}" ${!canEdit ? 'disabled' : ''}
                                   data-path="${JSON.stringify([...parentPath, key])}" data-type="path"
                                   class="param-input" style="flex:1;" />
                            <button class="btn btn-outline" style="padding:0 10px;height:30px;font-size:12px;" 
                                    onclick="browsePath(this)" ${!canEdit ? 'disabled' : ''}>📂</button>
                        </div>
                    `;
                    break;
                }
                case 'int[]':
                case 'float[]':
                case 'string[]': {
                    const arr = Array.isArray(value) ? value : [];
                    const isNum = type !== 'string[]';
                    controlHtml = `<div class="array-ctrl" data-path="${JSON.stringify([...parentPath, key])}" data-type="${type}">`;
                    arr.forEach((item, idx) => {
                        controlHtml += `
                            <span class="arr-item">
                                <input type="${isNum ? 'number' : 'text'}" value="${item}" ${!canEdit ? 'disabled' : ''}
                                       data-idx="${idx}" style="width:60px;" />
                                <span class="del-btn" onclick="removeArrayItem(this)" ${!canEdit ? 'style.display:none;' : ''}>✕</span>
                            </span>
                        `;
                    });
                    if (canEdit) controlHtml += `<button class="add-btn" onclick="addArrayItem(this)">＋</button>`;
                    controlHtml += `</div>`;
                    break;
                }
                default:
                    controlHtml = `<span class="text-muted">未知类型: ${type}</span>`;
            }
            return `
                <div class="param-item" data-path="${JSON.stringify([...parentPath, key])}">
                    <span class="p-name">${key}</span>
                    <div class="p-meta">
                        <span class="type-tag">${type}</span>
                        <span class="comp-tag ${compColor}">Lv${comp} · ${compLabel}</span>
                        ${!canEdit ? '<span style="color:var(--accent-red);font-size:11px;">🔒</span>' : ''}
                    </div>
                    <div class="p-control">${controlHtml}</div>
                </div>
            `;
        }

        // ---- 全局函数用于 onclick ----
        window.toggleSubGroup = function(el) {
            const sub = el.nextElementSibling;
            if (sub) {
                sub.style.display = sub.style.display === 'none' ? 'block' : 'none';
                const arrow = el.querySelector('.arrow-sub');
                if (arrow) arrow.textContent = sub.style.display === 'none' ? '▶' : '▼';
            }
        };
        window.toggleBool = function(trackEl) {
            const container = trackEl.closest('.bool-switch');
            if (!container) return;
            const isOn = trackEl.classList.contains('on');
            trackEl.classList.toggle('on');
            const label = container.querySelector('.bool-label');
            if (label) label.textContent = isOn ? 'FALSE' : 'TRUE';
            const path = JSON.parse(container.dataset.path || '[]');
            const newVal = isOn ? 'FALSE' : 'TRUE';
            setParamValue(path, newVal);
        };
        window.addArrayItem = function(btn) {
            const container = btn.closest('.array-ctrl');
            if (!container) return;
            const path = JSON.parse(container.dataset.path || '[]');
            const type = container.dataset.type || 'string[]';
            let arr = getParamValue(path) || [];
            if (!Array.isArray(arr)) arr = [];
            const newItem = type === 'int[]' ? 0 : type === 'float[]' ? 0.0 : '';
            arr.push(newItem);
            setParamValue(path, arr);
            showParamDetail(selectedPath);
        };
        window.removeArrayItem = function(el) {
            const item = el.closest('.arr-item');
            if (!item) return;
            const container = item.closest('.array-ctrl');
            if (!container) return;
            const path = JSON.parse(container.dataset.path || '[]');
            const idx = parseInt(item.querySelector('input')?.dataset.idx ?? 0);
            let arr = getParamValue(path) || [];
            if (!Array.isArray(arr)) return;
            arr.splice(idx, 1);
            setParamValue(path, arr);
            showParamDetail(selectedPath);
        };
        window.browsePath = function(btn) {
            const input = btn.closest('div').querySelector('input');
            if (!input) return;
            const fakePath = prompt('请输入路径:', input.value || './');
            if (fakePath !== null) {
                input.value = fakePath;
                const path = JSON.parse(input.dataset.path || '[]');
                setParamValue(path, fakePath);
            }
        };

        function getParamValue(path) {
            let obj = state.params;
            for (const key of path) {
                if (obj && obj[key] !== undefined) obj = obj[key];
                else return undefined;
            }
            return obj;
        }

        function setParamValue(path, value) {
            let obj = state.params;
            for (let i = 0; i < path.length - 1; i++) {
                const key = path[i];
                if (obj[key] === undefined) obj[key] = {};
                obj = obj[key];
            }
            const lastKey = path[path.length - 1];
            if (obj && typeof obj === 'object') {
                if (obj[lastKey] && typeof obj[lastKey] === 'object' && obj[lastKey].Type !== undefined) {
                    obj[lastKey].Value = value;
                } else {
                    obj[lastKey] = value;
                }
            }
        }

        // 监听输入变化（不记录日志）
        document.addEventListener('input', function(e) {
            const el = e.target;
            if (el.classList.contains('param-input') && !el.disabled) {
                const path = JSON.parse(el.dataset.path || '[]');
                if (path.length === 0) return;
                let value = el.value;
                const type = el.dataset.type || 'string';
                if (type === 'int') value = parseInt(value) || 0;
                else if (type === 'float') value = parseFloat(value) || 0.0;
                setParamValue(path, value);
            }
        });
        document.addEventListener('change', function(e) {
            const el = e.target;
            if (el.tagName === 'SELECT' && el.classList.contains('param-input') && !el.disabled) {
                const path = JSON.parse(el.dataset.path || '[]');
                if (path.length === 0) return;
                const value = el.value;
                setParamValue(path, value);
            }
        });

        // ---- 弹窗控制 ----
        function openModal(category) {
            if (!state.connected) {
                addBackendLog('⚠️ 未连接，无法打开参数');
                return;
            }
            state.currentCategory = category;
            modalTitle.textContent = category;
            renderTreeForCategory(category);
            modal.classList.add('active');
        }
        $('btnCloseParams').addEventListener('click', () => modal.classList.remove('active'));
        modal.addEventListener('click', (e) => { if (e.target === modal) modal.classList.remove('active'); });
        $('btnSaveParams').addEventListener('click', function() {
            addBackendLog(`参数 (${state.currentCategory}) 已保存`);
            const orig = this.textContent;
            this.textContent = '✅ 已保存';
            setTimeout(() => { this.textContent = orig; }, 1200);
        });

        // ================================================================
        //  9. 初始化
        // ================================================================
        resizeCanvas();
        addBackendLog('系统启动，正在自动连接...');
        autoConnect();

        // 加载图像按钮
        $('btnLoadImage').addEventListener('click', () => $('fileInput').click());
        $('fileInput').addEventListener('change', (e) => {
            if (e.target.files.length) loadImageFile(e.target.files[0]);
            e.target.value = '';
        });
        // 拖拽加载
        container.addEventListener('dragover', (e) => e.preventDefault());
        container.addEventListener('drop', (e) => {
            e.preventDefault();
            const files = e.dataTransfer.files;
            if (files.length && files[0].type.startsWith('image/')) loadImageFile(files[0]);
        });

        window.addEventListener('resize', resizeCanvas);
        const ro = new ResizeObserver(() => resizeCanvas());
        ro.observe(container);

        // 默认高亮算法
        const defaultTab = paramTabs.querySelector('[data-category="算法参数"]');
        if (defaultTab) defaultTab.classList.add('active');

        console.log('🔧 工业视觉检测系统 三列布局 (图像+日志统计+缺陷小图)');
    </script>
